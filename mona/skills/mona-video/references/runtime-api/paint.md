# PAINT: paint

Methods are available from createMotionToolkit / MotionCanvas kit. These are signatures from the adapted source, not executable code. P means PAINT; R means RIG. Use shared theme and actual speech timing.

```js
PAINT.canvas = (w = W, h = H) => ...
PAINT.cached = (key, w, h, fn) => ...
PAINT.scratch = (key) => ...
PAINT.noise = (x, y) => ...
PAINT.fbm = (x, y, o = 4) => ...
PAINT.hex = h => ...
PAINT.rgb = (c, a = 1) => ...
PAINT.mix = (a, b, t) => ...
PAINT.jitter = (c, r, amt) => ...
PAINT.texture = (key, base, opt = {}) => ...
PAINT.grain = (key, density = 0.08, col = [0, 0, 0], alpha = 0.18) => ...
PAINT.craquelure = (key, alpha = 0.35) => ...
PAINT.boilSeed = (t, fps = 12) => ...
PAINT.roughPath = (c, pts, { amp = 2, seed = 1, closed = false, step = 6 } = {}) => ...
PAINT.bez = (p0, p1, p2, p3, n = 24) => ...
PAINT.path = (d) => ...
PAINT.strokes = (dst, src, { cell = 12, len = 22, width = 7, angle, seed = 1, t = 0, boil = 8, outline = 0.35, outlineCol = [20, 25, 60], outlineMix = 0.55, jitterCol = 18, mask, alphaMask = false, palette, cap = 'round' } = {}) => ...
PAINT.swatch = (hexes, mixBase = 0.35) => ...
PAINT.dabs = (dst, src, { cell = 10, size = 9, seed = 2, t = 0, boil = 8, angle = -0.6, shift = 22 } = {}) => ...
PAINT.mosaic = (dst, src, { tile = 14, grout = '#3b352e', seed = 3, jitter = 0.25 } = {}) => ...
PAINT.halftone = (dst, src, { cell = 14, paper = '#f4ecd8', seed = 4 } = {}) => ...
PAINT.pixelate = (dst, src, { size = 8, palette } = {}) => ...
PAINT.facets = (dst, src, { nx = 22, ny = 13, seed = 5, t = 0, shiftAmt = 26, edge = 'rgba(40,30,20,.45)' } = {}) => ...
PAINT.cupCycle = (t, period = 2.4) => ...
PAINT.choreo = (lt, t) => ...
PAINT.steam = (c, t, x, y, { h = 80, n = 3, color = 'rgba(255,255,255,.8)', width = 4, spread = 16, wobble = 10, seed = 1 } = {}) => ...
PAINT.particles = (n, seed, t, { x0 = 0, x1 = W, y0 = -40, y1 = H + 40, speed = 80, drift = 30, life = 6 } = {}) => ...
```

Source: [upstream MIT library](https://github.com/alchaincyf/huashu-art-motion/blob/64cb27271ee0ad3375298e2890b33cbec36d4051/scripts/engine/lib/paint.js). Mona uses isolated instances; omit external engine globals, author presets and external resource loaders. Font binaries/stroke-order data are not bundled.
